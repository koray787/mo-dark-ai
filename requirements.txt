def db_connection():
    conn = sqlite3.connect(
        DB_FILE,
        check_same_thread=False,
    )
    conn.row_factory = sqlite3.Row
    return conn

def init_database():
    conn = db_connection()
    # Enhanced sessions table with metadata
    conn.execute(
        """ CREATE TABLE IF NOT EXISTS sessions (
            id TEXT PRIMARY KEY,
            title TEXT NOT NULL,
            created_at TEXT NOT NULL,
            updated_at TEXT NOT NULL,
            model_used TEXT,
            total_messages INTEGER DEFAULT 0,
            tags TEXT,
            is_favorite INTEGER DEFAULT 0
        ) """
    )
    # Enhanced messages table with additional fields
    conn.execute(
        """ CREATE TABLE IF NOT EXISTS messages (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            session_id TEXT NOT NULL,
            role TEXT NOT NULL,
            content TEXT NOT NULL,
            content_type TEXT DEFAULT 'text',
            created_at TEXT NOT NULL,
            tokens_used INTEGER DEFAULT 0,
            model_used TEXT,
            media_url TEXT
        ) """
    )
    # New table for generated media
    conn.execute(
        """ CREATE TABLE IF NOT EXISTS generated_media (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            session_id TEXT NOT NULL,
            media_type TEXT NOT NULL,
            prompt TEXT NOT NULL,
            media_data TEXT NOT NULL,
            created_at TEXT NOT NULL
        ) """
    )
    # New table for user preferences
    conn.execute(
        """ CREATE TABLE IF NOT EXISTS user_preferences (
            id INTEGER PRIMARY KEY CHECK (id = 1),
            theme TEXT DEFAULT 'dark',
            language TEXT DEFAULT 'ar',
            auto_save INTEGER DEFAULT 1,
            max_tokens INTEGER DEFAULT 2048,
            model_preferences TEXT
        ) """
    )
    # Insert default preferences if not exists
    conn.execute(
        """ INSERT OR IGNORE INTO user_preferences (id) VALUES (1) """
    )
    conn.commit()
    conn.close()

def create_session(title="محادثة جديدة"):
    session_id = str(uuid.uuid4())
    now = datetime.now().isoformat(timespec="seconds")
    conn = db_connection()
    conn.execute(
        """ INSERT INTO sessions (id, title, created_at, updated_at, model_used, total_messages)
        VALUES (?, ?, ?, ?, ?, ?) """,
        (
            session_id,
            title,
            now,
            now,
            TEXT_MODEL,
            0,
        ),
    )
    conn.commit()
    conn.close()
    return session_id

def update_session_title(session_id, title):
    title = title.replace("\n", " ").strip()
    if not title:
        title = "محادثة جديدة"
    title = title[:80]
    conn = db_connection()
    conn.execute(
        """ UPDATE sessions SET title = ?, updated_at = ? WHERE id = ? """,

        ),
    )
    conn.commit()
    conn.close()

def touch_session(session_id):
    conn = db_connection()
    conn.execute(
        """ UPDATE sessions SET updated_at = ? WHERE id = ? """,
        (
            datetime.now().isoformat(timespec="seconds"),
            session_id,
        ),
    )
    conn.commit()
    conn.close()

def save_message(session_id, role, content, content_type="text", tokens_used=0, model_used=None, media_url=None):
    conn = db_connection()
    conn.execute(
        """ INSERT INTO messages (session_id, role, content, content_type, created_at, tokens_used, model_used, media_url)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?) """,
        (
            session_id,
            role,
            content,
            content_type,
            datetime.now().isoformat(timespec="seconds"),
            tokens_used,
            model_used,
            media_url,
        ),
    )
    conn.commit()
    
    # Update message count in sessions
    conn.execute(
        """ UPDATE sessions SET total_messages = total_messages + 1 WHERE id = ? """,
        (session_id,)
    )
    conn.commit()
    conn.close()
    touch_session(session_id)

def update_user_preferences(theme=None, language=None, auto_save=None, max_tokens=None):
    conn = db_connection()
    updates = []
    params = []
    
    if theme is not None:
        updates.append("theme = ?")
        params.append(theme)
    
    if language is not None:
        updates.append("language = ?")
        params.append(language)
    
    if auto_save is not None:
        updates.append("auto_save = ?")
        params.append(auto_save)
    
    if max_tokens is not None:
        updates.append("max_tokens = ?")
        params.append(max_tokens)
    
    if updates:
        params.append(1)  # id = 1
        conn.execute(
            f""" UPDATE user_preferences SET {', '.join(updates)} WHERE id = ? """,
            params
        )
        conn.commit()
    
    conn.close()

def get_session_stats():
    conn = db_connection()
    stats = conn.execute(
        """ SELECT 
            COUNT(*) as total_sessions,
            COUNT(CASE WHEN is_favorite = 1 THEN 1 END) as favorite_sessions,
            AVG(total_messages) as avg_messages_per_session,
            MAX(updated_at) as last_activity
        FROM sessions """
    ).fetchone()
    conn.close()
    return stats
